The logic remains rock-solid. Ensure your Wemos is wired correctly: Fans (D1, D5, D2) and Relays (D0, D3, D4, D6, D7, D8).
#include <ESP8266WiFi.h>
#include <ESP8266WebServer.h>

const char* ssid = "Wemos_Home_Auto";
const char* password = "87651234"; 

ESP8266WebServer server(80);

// Pins & States for 3 Fans
const int fanPins[3] = {D1, D5, D2}; 
int fanSpeeds[3] = {0, 0, 0}; 
bool fanStates[3] = {false, false, false}; 

// Pins & States for 6 Relays
const int relayPins[6] = {D0, D3, D4, D6, D7, D8};
bool relayStates[6] = {false, false, false, false, false, false}; 

void setup() {
  Serial.begin(115200);

  for (int i = 0; i < 3; i++) {
    pinMode(fanPins[i], OUTPUT);
    analogWrite(fanPins[i], 1023); // 1023 = Active-LOW OFF
  }

  for (int i = 0; i < 6; i++) {
    pinMode(relayPins[i], OUTPUT);
    digitalWrite(relayPins[i], HIGH); // HIGH = Active-LOW OFF
  }

  WiFi.softAP(ssid, password);
  Serial.print("Hotspot Started. AP IP: ");
  Serial.println(WiFi.softAPIP());

  server.on("/relay", HTTP_GET, handleRelay);
  server.on("/fan_state", HTTP_GET, handleFanState);
  server.on("/fan_speed", HTTP_GET, handleFanSpeed);
  server.on("/status", HTTP_GET, handleStatus);
  server.begin();
}

void loop() {
  server.handleClient();
}

void updateFanHardware(int id) {
  if (!fanStates[id]) {
    analogWrite(fanPins[id], 1023); 
  } else {
    int pwmValue = map(fanSpeeds[id], 0, 100, 1023, 0); 
    analogWrite(fanPins[id], pwmValue);
  }
}

void handleRelay() {
  if (server.hasArg("id") && server.hasArg("state")) {
    int id = server.arg("id").toInt(); 
    String state = server.arg("state");

    if (id >= 0 && id < 6) {
      if (state == "on") {
        digitalWrite(relayPins[id], LOW); 
        relayStates[id] = true;
      } else {
        digitalWrite(relayPins[id], HIGH); 
        relayStates[id] = false;
      }
      server.send(200, "text/plain", "OK");
      return;
    }
  }
  server.send(400, "text/plain", "Bad Request");
}

void handleFanState() {
  if (server.hasArg("id") && server.hasArg("state")) {
    int id = server.arg("id").toInt(); 
    String state = server.arg("state");

    if (id >= 0 && id < 3) {
      fanStates[id] = (state == "on");
      updateFanHardware(id);
      server.send(200, "text/plain", "OK");
      return;
    }
  }
  server.send(400, "text/plain", "Bad Request");
}

void handleFanSpeed() {
  if (server.hasArg("id") && server.hasArg("speed")) {
    int id = server.arg("id").toInt(); 
    int speed = server.arg("speed").toInt(); 

    if (id >= 0 && id < 3 && speed >= 0 && speed <= 100) {
      fanSpeeds[id] = speed;
      updateFanHardware(id);
      server.send(200, "text/plain", "OK");
      return;
    }
  }
  server.send(400, "text/plain", "Bad Request");
}

void handleStatus() {
  String json = "{";
  json += "\"r\":[";
  for(int i = 0; i < 6; i++) {
    json += (relayStates[i] ? "true" : "false");
    if(i < 5) json += ",";
  }
  json += "],\"fs\":[";
  for(int i = 0; i < 3; i++) {
    json += (fanStates[i] ? "true" : "false");
    if(i < 2) json += ",";
  }
  json += "],\"f\":[";
  json += fanSpeeds[0]; json += ","; json += fanSpeeds[1]; json += ","; json += fanSpeeds[2];
  json += "]}";
  
  server.send(200, "application/json", json);
}

package com.watari.homepulse

import android.animation.ObjectAnimator
import android.content.Context
import android.content.SharedPreferences
import android.graphics.Color
import android.os.Bundle
import android.view.Gravity
import android.view.View
import android.view.animation.LinearInterpolator
import android.widget.*
import androidx.appcompat.app.AppCompatActivity
import com.google.android.material.bottomnavigation.BottomNavigationView
import org.json.JSONObject
import java.io.BufferedReader
import java.io.InputStreamReader
import java.net.HttpURLConnection
import java.net.URL
import java.util.concurrent.Executors

class MainActivity : AppCompatActivity() {

    private val executor = Executors.newSingleThreadExecutor()
    private lateinit var prefs: SharedPreferences
    private var currentIp = "192.168.4.1"

    private val relayStates = BooleanArray(6) { false }
    private val fanStates = BooleanArray(3) { false }
    private val fanSpeeds = IntArray(3) { 0 }

    private val relaySwitches = ArrayList<Switch>()
    private val relayImages = ArrayList<ImageView>()
    private val fanSwitches = ArrayList<Switch>()
    private val fanImages = ArrayList<ImageView>()
    private val fanSeekBars = ArrayList<SeekBar>()
    private val fanAnimators = ArrayList<ObjectAnimator>()

    private val relayNames = arrayOf("Socket 1 (D0)", "Tubelight 1 (D3)", "Socket 2 (D4)", "Bulb 1 (D6)", "Tubelight 2 (D7)", "Socket 3 (D8)")
    private val relayIcons = arrayOf(R.drawable.ic_socket, R.drawable.ic_tube, R.drawable.ic_socket, R.drawable.ic_bulb, R.drawable.ic_tube, R.drawable.ic_socket)
    private val fanNames = arrayOf("Fan 1 (D1)", "Fan 2 (D5)", "Fan 3 (D2)")

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        setContentView(R.layout.activity_main)

        prefs = getSharedPreferences("HomepulsePrefs", Context.MODE_PRIVATE)
        currentIp = prefs.getString("WEMOS_IP", "192.168.4.1") ?: "192.168.4.1"

        setupNavigationAndSettings()
        generateDashboardUI()
    }

    override fun onResume() {
        super.onResume()
        fetchHardwareStatus()
    }

    private fun generateDashboardUI() {
        val container = findViewById<LinearLayout>(R.id.hardwareContainer)
        val cardBg = Color.parseColor("#D9121212")

        for (i in 0 until 3) {
            val card = LinearLayout(this).apply {
                orientation = LinearLayout.HORIZONTAL
                setBackgroundColor(cardBg)
                setPadding(30, 30, 30, 30)
                gravity = Gravity.CENTER_VERTICAL
                layoutParams = LinearLayout.LayoutParams(LinearLayout.LayoutParams.MATCH_PARENT, LinearLayout.LayoutParams.WRAP_CONTENT).apply { setMargins(0,0,0,24) }
            }

            val img = ImageView(this).apply {
                setImageResource(R.drawable.ic_fan)
                setColorFilter(Color.parseColor("#555555"))
                layoutParams = LinearLayout.LayoutParams(160, 160)
            }
            fanImages.add(img)

            val anim = ObjectAnimator.ofFloat(img, "rotation", 0f, 360f).apply {
                repeatCount = ObjectAnimator.INFINITE
                interpolator = LinearInterpolator()
            }
            fanAnimators.add(anim)

            val controlBox = LinearLayout(this).apply {
                orientation = LinearLayout.VERTICAL
                setPadding(30, 0, 0, 0)
                layoutParams = LinearLayout.LayoutParams(0, LinearLayout.LayoutParams.WRAP_CONTENT, 1f)
            }

            val sw = Switch(this).apply {
                text = fanNames[i]
                setTextColor(Color.WHITE)
                textSize = 18f
                setOnClickListener { toggleFanState(i) }
            }
            fanSwitches.add(sw)

            val seek = SeekBar(this).apply {
                max = 100
                isEnabled = false
                setPadding(0, 20, 0, 0)
                setOnSeekBarChangeListener(object : SeekBar.OnSeekBarChangeListener {
                    override fun onProgressChanged(seekBar: SeekBar?, progress: Int, fromUser: Boolean) {
                        if (fromUser && fanStates[i]) setFanSpeed(i, progress)
                    }
                    override fun onStartTrackingTouch(seekBar: SeekBar?) {}
                    override fun onStopTrackingTouch(seekBar: SeekBar?) {}
                })
            }
            fanSeekBars.add(seek)

            controlBox.addView(sw)
            controlBox.addView(seek)
            card.addView(img)
            card.addView(controlBox)
            container.addView(card)
        }

        for (i in 0 until 6) {
            val card = LinearLayout(this).apply {
                orientation = LinearLayout.HORIZONTAL
                setBackgroundColor(cardBg)
                setPadding(30, 30, 30, 30)
                gravity = Gravity.CENTER_VERTICAL
                layoutParams = LinearLayout.LayoutParams(LinearLayout.LayoutParams.MATCH_PARENT, LinearLayout.LayoutParams.WRAP_CONTENT).apply { setMargins(0,0,0,24) }
            }

            val img = ImageView(this).apply {
                setImageResource(relayIcons[i])
                setColorFilter(Color.parseColor("#555555"))
                layoutParams = LinearLayout.LayoutParams(160, 160)
            }
            relayImages.add(img)

            val sw = Switch(this).apply {
                text = relayNames[i]
                setTextColor(Color.WHITE)
                textSize = 18f
                setPadding(30, 0, 0, 0)
                layoutParams = LinearLayout.LayoutParams(0, LinearLayout.LayoutParams.WRAP_CONTENT, 1f)
                setOnClickListener { toggleRelay(i) }
            }
            relaySwitches.add(sw)

            card.addView(img)
            card.addView(sw)
            container.addView(card)
        }
    }

    private fun updateVisuals() {
        for (i in 0 until 6) {
            relaySwitches[i].isChecked = relayStates[i]
            if (relayStates[i]) {
                val tint = if (relayNames[i].contains("Socket")) "#00E676" else "#FFEB3B"
                relayImages[i].setColorFilter(Color.parseColor(tint))
            } else {
                relayImages[i].setColorFilter(Color.parseColor("#555555"))
            }
        }

        for (i in 0 until 3) {
            fanSwitches[i].isChecked = fanStates[i]
            fanSeekBars[i].isEnabled = fanStates[i]
            fanSeekBars[i].progress = fanSpeeds[i]
            
            if (fanStates[i]) {
                fanImages[i].setColorFilter(Color.parseColor("#00E676"))
                if (fanSpeeds[i] > 0) {
                    val duration = ((100 - fanSpeeds[i]) * 15) + 300
                    fanAnimators[i].duration = duration.toLong()
                    if (!fanAnimators[i].isRunning) fanAnimators[i].start()
                } else {
                    fanAnimators[i].cancel()
                }
            } else {
                fanImages[i].setColorFilter(Color.parseColor("#555555"))
                fanAnimators[i].cancel()
            }
        }
    }

    private fun fetchHardwareStatus() {
        executor.execute {
            try {
                val url = URL("http://$currentIp/status")
                val conn = url.openConnection() as HttpURLConnection
                conn.requestMethod = "GET"
                conn.connectTimeout = 2000

                val reader = BufferedReader(InputStreamReader(conn.inputStream))
                val response = reader.readText()
                reader.close()
                conn.disconnect()

                val json = JSONObject(response)
                val rArray = json.getJSONArray("r")
                val fsArray = json.getJSONArray("fs")
                val fArray = json.getJSONArray("f")

                runOnUiThread {
                    for (i in 0 until 6) relayStates[i] = rArray.getBoolean(i)
                    for (i in 0 until 3) { 
                        fanStates[i] = fsArray.getBoolean(i)
                        fanSpeeds[i] = fArray.getInt(i)
                    }
                    updateVisuals()
                }
            } catch (e: Exception) {}
        }
    }

    private fun toggleRelay(i: Int) {
        val target = !relayStates[i]
        executor.execute {
            try {
                val conn = URL("http://$currentIp/relay?id=$i&state=${if (target) "on" else "off"}").openConnection() as HttpURLConnection
                conn.requestMethod = "GET"
                conn.connectTimeout = 1000
                if (conn.responseCode == 200) runOnUiThread { relayStates[i] = target; updateVisuals() }
                conn.disconnect()
            } catch (e: Exception) {}
        }
    }

    private fun toggleFanState(i: Int) {
        val target = !fanStates[i]
        executor.execute {
            try {
                val conn = URL("http://$currentIp/fan_state?id=$i&state=${if (target) "on" else "off"}").openConnection() as HttpURLConnection
                conn.requestMethod = "GET"
                conn.connectTimeout = 1000
                if (conn.responseCode == 200) runOnUiThread { fanStates[i] = target; updateVisuals() }
                conn.disconnect()
            } catch (e: Exception) {}
        }
    }

    private fun setFanSpeed(i: Int, speed: Int) {
        fanSpeeds[i] = speed
        updateVisuals()
        executor.execute {
            try {
                val conn = URL("http://$currentIp/fan_speed?id=$i&speed=$speed").openConnection() as HttpURLConnection
                conn.requestMethod = "GET"
                conn.connectTimeout = 500
                conn.responseCode
                conn.disconnect()
            } catch (e: Exception) {}
        }
    }

    private fun setupNavigationAndSettings() {
        val flipper = findViewById<ViewFlipper>(R.id.viewFlipper)
        findViewById<BottomNavigationView>(R.id.bottomNav).setOnItemSelectedListener { item ->
            if (item.itemId == R.id.nav_home) flipper.displayedChild = 0 else flipper.displayedChild = 1
            true
        }
        val pass = findViewById<EditText>(R.id.inputPassword)
        val unlock = findViewById<Button>(R.id.btnUnlock)
        val hidden = findViewById<LinearLayout>(R.id.hiddenIpLayout)
        val ip = findViewById<EditText>(R.id.inputIp).apply { setText(currentIp) }
        
        unlock.setOnClickListener {
            if (pass.text.toString() == "1234qwer") {
                pass.visibility = View.GONE; unlock.visibility = View.GONE; hidden.visibility = View.VISIBLE
            } else Toast.makeText(this, "Incorrect Password", Toast.LENGTH_SHORT).show()
        }
        findViewById<Button>(R.id.btnSaveSettings).setOnClickListener {
            if (ip.text.toString().isNotEmpty()) {
                currentIp = ip.text.toString().trim()
                prefs.edit().putString("WEMOS_IP", currentIp).apply()
                Toast.makeText(this, "IP Saved", Toast.LENGTH_SHORT).show()
                fetchHardwareStatus()
            }
        }
    }
}
